uint8_t get_gsm_tcp_status(int time_to_wait) {

	 uint8_t delim_postns[20] = {0};
	 uint8_t pstn[20];
	 uint8_t stat_max = 64; //max value for array was 32
	 uint8_t _stat[stat_max];
	 char *vr;

	 readDataBufPtr = 0; //reset the bufferpointer
	 memset(&gsm_data_buf[0], 0, Read_data_MAX-1);
	 gprs_uart6_send((uint8_t *)"AT+CIPSTATUS \r",sizeof("AT+CIPSTATUS \r"));

	state = satateIdle;

	gsm_ticks = 0; //reset the timer
	while (gsm_ticks < time_to_wait){  //Do this before timer expires


		while(gprs_uart6_test()){
			if(readDataBufPtr < Read_data_MAX) {
				gsm_data_buf[readDataBufPtr++] = gprs_uart6_getc();
			}else readDataBufPtr = 0;
			state = satateMessageReceived;
			delay(10);
		}

		if (state == satateMessageReceived){
			//split the result data
			split_str2_delim((uint8_t *)gsm_data_buf,'\r','\n',Read_data_MAX,delim_postns);

			read_str_from_str((uint8_t *)gsm_data_buf,_stat,stat_max,(delim_postns[2]),(delim_postns[3]));


			if(strstr((char *)_stat,"IP INTIAL")){
				ip_conn_state = IP_INITIAL;
			}else if(strstr((char *)_stat,"IP START")){
				ip_conn_state = IP_START;
			}else if(strstr((char *)_stat,"IP CONFIG")){
				ip_conn_state = IP_CONFIG;
			}else if(strstr((char *)_stat,"IP GPRSACT")){
				ip_conn_state = IP_GPRSACT;
			}else if(strstr((char *)_stat,"IP STATUS")){
				ip_conn_state = IP_STATUS;
			}else if(strstr((char *)_stat,"IP PROCESSING")){
				ip_conn_state = IP_PROCESSING;
			}else if(strstr((char *)_stat,"PDP DEACT")){
				ip_conn_state = PDP_DEACT;
			}

			//Check status of connection 1

			//Check connection 1 delim_postns[6] split further.
			vr = &gsm_data_buf[delim_postns[4]];
			split_str((uint8_t *)vr, ',',delim_postns[5]-delim_postns[4], pstn);
			read_str_from_str((uint8_t *)gsm_data_buf,_stat,stat_max,(delim_postns[4]+pstn[4]),(delim_postns[5]));
			if(strstr((char *)_stat,"INITIAL")){
				main_tcp_conn.state = INITIAL;
			}else if(strstr((char *)_stat,"CONNECTING")){
				main_tcp_conn.state = CONNECTING;
			}else if(strstr((char *)_stat,"CONNECTED")){
				main_tcp_conn.state = CONNECTED;
			}else if(strstr((char *)_stat,"REMOTE CLOSING")){
				main_tcp_conn.state = REMOTE_CLOSING;
			}else if(strstr((char *)_stat,"CLOSING")){
				main_tcp_conn.state = CLOSING;
			}else if(strstr((char *)_stat,"CLOSED")){
				main_tcp_conn.state = CLOSED;
			}

			//update the connection status register with the result
			state = satateIdle;
			//string manipulations



		}

	}
	readDataBufPtr = 0;
	return 0;

}
